cmake_minimum_required(VERSION 3.22)
project(gsim_standalone VERSION 0.17.0 LANGUAGES CXX)
include(GNUInstallDirs)
add_library(gsim_workflow STATIC src/simulation_workflow.cpp src/gaussian_workflow.cpp src/packed_chromosome.cpp src/bed_storage.cpp src/metadata_storage.cpp)
target_compile_features(gsim_workflow PUBLIC cxx_std_17)
target_include_directories(gsim_workflow PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
add_executable(gsim_native tools/standalone/simulation_main.cpp)
target_link_libraries(gsim_native PRIVATE gsim_workflow)
install(TARGETS gsim_native RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(PROGRAMS tools/standalone/gsim DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES tools/standalone/gsim.py DESTINATION ${CMAKE_INSTALL_DATADIR}/gsim)
